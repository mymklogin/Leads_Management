[HttpPost("GetRoutingInfo")]
public async Task<IActionResult> HandleTtsSimpleIvrWebhook(
    [FromBody] ExpressIvrWebhookDto payload,
    CancellationToken cancellationToken)
{
    if (payload == null)
    {
        return BadRequest(new
        {
            status = "error",
            message = "Invalid webhook payload."
        });
    }

    string eventType =
        payload.EventType?.Trim().ToUpperInvariant() ?? "";

    string pressedKey =
        payload.PressedKey?.Trim() ?? "";

    int duration = Math.Max(0, payload.Duration);

    if (string.IsNullOrWhiteSpace(eventType))
    {
        return BadRequest(new
        {
            status = "error",
            message = "Event type is required."
        });
    }

    if (string.IsNullOrWhiteSpace(payload.Mobile))
    {
        return BadRequest(new
        {
            status = "error",
            message = "Mobile number is required."
        });
    }

    // ---------------------------------------------------------
    // Temporary DTMF cache key
    // ---------------------------------------------------------

    string cacheKey = $"TTS_SIMPLE_DTMF_{payload.Mobile}";

    string? temporaryDtmf = null;

    // =========================================================
    // CONNECTED CALL
    // =========================================================

    if (eventType == "CONNECTED_CALLS")
    {
        return Ok(new
        {
            status = "success",
            templateId = 7,
            eventType,
            pressedKey,
            duration,
            message = "TTS Simple IVR call connected."
        });
    }

    // =========================================================
    // DTMF
    //
    // Do NOT save into DB here.
    // Temporarily hold actual PressedKey.
    // =========================================================

    if (eventType == "DTMF")
    {
        if (!string.IsNullOrWhiteSpace(pressedKey))
        {
            _cache.Set(
                cacheKey,
                pressedKey,
                TimeSpan.FromMinutes(30));
        }

        return Ok(new
        {
            status = "success",
            templateId = 7,
            eventType,
            pressedKey,
            duration,
            message = "TTS DTMF temporarily stored."
        });
    }

    // =========================================================
    // Only HANGUP performs final DB save
    // =========================================================

    if (eventType != "HANGUP")
    {
        return BadRequest(new
        {
            status = "error",
            message =
                $"Unsupported event type: {payload.EventType}"
        });
    }

    // =========================================================
    // HANGUP
    //
    // First priority:
    // HANGUP PressedKey, if provider sends it.
    //
    // Otherwise:
    // DTMF temporarily stored in cache.
    // =========================================================

    if (!string.IsNullOrWhiteSpace(pressedKey))
    {
        temporaryDtmf = pressedKey;
    }
    else
    {
        _cache.TryGetValue(
            cacheKey,
            out temporaryDtmf);
    }

    temporaryDtmf =
        temporaryDtmf?.Trim() ?? "";

    // =========================================================
    // FINAL LEAD STATUS
    // =========================================================

    string leadStatus;

    if (temporaryDtmf == "1" && duration >= 60)
    {
        leadStatus =
            "Very Hot Lead - TTS Target Key 1 + Deep Engagement";
    }
    else if (temporaryDtmf == "1" && duration >= 30)
    {
        leadStatus =
            "Hot Lead - TTS Target Key 1 + Good Engagement";
    }
    else if (temporaryDtmf == "1")
    {
        leadStatus =
            "Interested Lead - TTS Target Key 1 Pressed";
    }
    else if (!string.IsNullOrWhiteSpace(temporaryDtmf) &&
             duration >= 60)
    {
        leadStatus =
            $"Hot Lead - Long TTS Engagement - Key {temporaryDtmf}";
    }
    else if (!string.IsNullOrWhiteSpace(temporaryDtmf) &&
             duration >= 30)
    {
        leadStatus =
            $"Warm Lead - TTS Engagement - Key {temporaryDtmf}";
    }
    else if (duration >= 60)
    {
        leadStatus =
            "Hot Lead - Long TTS Engagement";
    }
    else if (duration >= 30)
    {
        leadStatus =
            "Warm Lead - Good TTS Engagement";
    }
    else if (duration >= 15)
    {
        leadStatus =
            "Cold Lead - Short TTS Engagement";
    }
    else if (duration >= 10)
    {
        leadStatus =
            "Quick TTS Interaction";
    }
    else if (duration >= 5)
    {
        leadStatus =
            "Intro Drop";
    }
    else
    {
        leadStatus =
            "Immediate Drop";
    }

    // =========================================================
    // FIND EXISTING LEAD
    //
    // DB lookup happens only at HANGUP.
    // =========================================================

    var lead = await _context.LeadRecords
        .FirstOrDefaultAsync(
            x => x.Mobile == payload.Mobile,
            cancellationToken);

    // =========================================================
    // FINAL DB SAVE
    // =========================================================

    if (lead == null)
    {
        lead = new LeadRecord
        {
            Mobile = payload.Mobile!,
            LeadStatus = leadStatus,
            CallDuration = duration,

            PressedDtmf =
                string.IsNullOrWhiteSpace(temporaryDtmf)
                    ? null
                    : temporaryDtmf,

            LastEventType = "HANGUP",
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };

        _context.LeadRecords.Add(lead);
    }
    else
    {
        lead.LeadStatus = leadStatus;
        lead.CallDuration = duration;
        lead.LastEventType = "HANGUP";
        lead.UpdatedAt = DateTime.UtcNow;

        lead.PressedDtmf =
            string.IsNullOrWhiteSpace(temporaryDtmf)
                ? null
                : temporaryDtmf;
    }

    // =========================================================
    // SAVE FINAL DATA
    // =========================================================

    await _context.SaveChangesAsync(cancellationToken);

    // =========================================================
    // REMOVE TEMPORARY DTMF
    // =========================================================

    _cache.Remove(cacheKey);

    // =========================================================
    // RESPONSE
    // =========================================================

    return Ok(new
    {
        status = "success",
        templateId = 7,
        eventType,
        pressedKey,
        effectivePressedKey = temporaryDtmf,
        duration,
        leadStatus,
        message = "HANGUP received. Final TTS lead data saved."
    });
}
