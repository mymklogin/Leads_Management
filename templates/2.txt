[HttpPost("GetRoutingInfo")]
public async Task<IActionResult> HandleCallPatchWebhook(
    [FromBody] ExpressIvrWebhookDto payload,
    CancellationToken cancellationToken)
{
    // ---------------------------------------------------------
    // Validate Payload
    // ---------------------------------------------------------

    if (payload == null)
    {
        return BadRequest(new
        {
            status = "error",
            message = "Invalid webhook payload."
        });
    }

    string eventType =
        payload.EventType?.Trim().ToUpperInvariant() ?? "";

    string pressedKey =
        payload.PressedKey?.Trim() ?? "";

    int duration = Math.Max(0, payload.Duration);

    if (string.IsNullOrWhiteSpace(eventType))
    {
        return BadRequest(new
        {
            status = "error",
            message = "Event type is required."
        });
    }

    // ---------------------------------------------------------
    // Find Existing Lead
    // ---------------------------------------------------------

    var lead = await _context.LeadRecords
        .FirstOrDefaultAsync(
            x => x.Mobile == payload.Mobile,
            cancellationToken);

    string leadStatus;

    // ---------------------------------------------------------
    // EVENT LOGIC
    // ---------------------------------------------------------

    switch (eventType)
    {
        // -----------------------------------------------------
        // CALL CONNECTED
        // -----------------------------------------------------

        case "CONNECTED_CALLS":

            leadStatus =
                "Call Patch Campaign Connected";

            break;


        // -----------------------------------------------------
        // DTMF
        // -----------------------------------------------------

        case "DTMF":

            if (pressedKey == "1")
            {
                leadStatus =
                    "Agent Connection Requested - Pressed 1";
            }
            else if (!string.IsNullOrWhiteSpace(pressedKey))
            {
                leadStatus =
                    $"Call Patch Interaction - Other Key Pressed ({pressedKey})";
            }
            else
            {
                leadStatus =
                    "DTMF Event - No Key Captured";
            }

            break;


        // -----------------------------------------------------
        // HANGUP
        // -----------------------------------------------------

        case "HANGUP":

            // IMPORTANT:
            // HANGUP ke time sirf actual HANGUP pressedKey
            // use kiya ja raha hai.
            //
            // DB se PressedDtmf fallback nahi liya ja raha.

            if (pressedKey == "1" && duration > 60)
            {
                leadStatus =
                    "Very Hot Lead - Agent Connection & Long Conversation";
            }
            else if (pressedKey == "1" && duration >= 30)
            {
                leadStatus =
                    "Hot Lead - Agent Connection & Good Conversation";
            }
            else if (pressedKey == "1")
            {
                leadStatus =
                    "Interested Lead - Requested Agent Connection";
            }
            else if (duration >= 60)
            {
                leadStatus =
                    "Hot Lead - Long Conversation Without Agent Request";
            }
            else if (duration >= 30)
            {
                leadStatus =
                    "Warm Lead - Good Call Engagement";
            }
            else if (duration >= 15)
            {
                leadStatus =
                    "Cold Lead - Short Call Engagement";
            }
            else if (duration >= 10)
            {
                leadStatus =
                    "Quick Interaction";
            }
            else if (duration >= 5)
            {
                leadStatus =
                    "Intro Drop";
            }
            else
            {
                leadStatus =
                    "Immediate Drop";
            }

            break;


        // -----------------------------------------------------
        // Unsupported Event
        // -----------------------------------------------------

        default:

            return BadRequest(new
            {
                status = "error",
                message =
                    $"Unsupported event type: {payload.EventType}"
            });
    }

    // ---------------------------------------------------------
    // CREATE / UPDATE LEAD
    // ---------------------------------------------------------

    if (lead == null)
    {
        lead = new LeadRecord
        {
            Mobile = payload.Mobile!,
            LeadStatus = leadStatus,
            CallDuration = duration,

            PressedDtmf =
                !string.IsNullOrWhiteSpace(pressedKey)
                    ? pressedKey
                    : null,

            LastEventType = eventType,
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };

        _context.LeadRecords.Add(lead);
    }
    else
    {
        lead.LeadStatus = leadStatus;
        lead.LastEventType = eventType;
        lead.UpdatedAt = DateTime.UtcNow;

        // -----------------------------------------------------
        // DTMF Event
        // -----------------------------------------------------

        if (eventType == "DTMF" &&
            !string.IsNullOrWhiteSpace(pressedKey))
        {
            lead.PressedDtmf = pressedKey;
        }

        // -----------------------------------------------------
        // HANGUP
        // -----------------------------------------------------

        if (eventType == "HANGUP")
        {
            lead.CallDuration = duration;

            // Sirf HANGUP ke actual pressedKey ko save karo.
            // Blank hai to purana DTMF overwrite nahi hoga.
            if (!string.IsNullOrWhiteSpace(pressedKey))
            {
                lead.PressedDtmf = pressedKey;
            }
        }
    }

    // ---------------------------------------------------------
    // Save
    // ---------------------------------------------------------

    await _context.SaveChangesAsync(cancellationToken);

    // ---------------------------------------------------------
    // Response
    // ---------------------------------------------------------

    return Ok(new
    {
        status = "success",
        templateId = 2,
        eventType,
        pressedKey,
        duration,
        leadStatus
    });
}
