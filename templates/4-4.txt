```csharp
[HttpPost("GetRoutingInfo")]
public async Task<IActionResult> HandleMultiLevelNextarWebhook(
    [FromBody] ExpressIvrWebhookDto payload,
    CancellationToken cancellationToken)
{
    if (payload == null)
    {
        return BadRequest(new
        {
            status = "error",
            message = "Invalid webhook payload."
        });
    }

    string eventType =
        payload.EventType?.Trim().ToUpperInvariant() ?? "";

    string pressedKey =
        payload.PressedKey?.Trim() ?? "";

    int duration = Math.Max(0, payload.Duration);

    if (string.IsNullOrWhiteSpace(eventType))
    {
        return BadRequest(new
        {
            status = "error",
            message = "Event type is required."
        });
    }

    if (string.IsNullOrWhiteSpace(payload.Mobile))
    {
        return BadRequest(new
        {
            status = "error",
            message = "Mobile number is required."
        });
    }

    string cacheKey = $"IVR_DTMF_{payload.Mobile}";

    string? temporaryDtmf = null;

    // ---------------------------------------------------------
    // CONNECTED CALL
    // ---------------------------------------------------------

    if (eventType == "CONNECTED_CALLS")
    {
        return Ok(new
        {
            status = "success",
            templateId = 4,
            eventType,
            pressedKey,
            duration,
            message = "Call connected."
        });
    }

    // ---------------------------------------------------------
    // DTMF
    //
    // IMPORTANT:
    // DTMF yahan DB me save nahi hoga.
    // Sirf temporary cache me hold hoga.
    // ---------------------------------------------------------

    if (eventType == "DTMF")
    {
        if (!string.IsNullOrWhiteSpace(pressedKey))
        {
            _cache.Set(
                cacheKey,
                pressedKey,
                TimeSpan.FromMinutes(30));
        }

        return Ok(new
        {
            status = "success",
            templateId = 4,
            eventType,
            pressedKey,
            duration,
            message = "DTMF received and temporarily stored."
        });
    }

    // ---------------------------------------------------------
    // Only HANGUP will create/update final DB record
    // ---------------------------------------------------------

    if (eventType != "HANGUP")
    {
        return BadRequest(new
        {
            status = "error",
            message =
                $"Unsupported event type: {payload.EventType}"
        });
    }

    // ---------------------------------------------------------
    // HANGUP
    //
    // First priority:
    // If HANGUP itself has PressedKey, use it.
    //
    // Otherwise:
    // Use DTMF temporarily stored during DTMF event.
    // ---------------------------------------------------------

    if (!string.IsNullOrWhiteSpace(pressedKey))
    {
        temporaryDtmf = pressedKey;
    }
    else
    {
        _cache.TryGetValue(
            cacheKey,
            out temporaryDtmf);
    }

    temporaryDtmf =
        temporaryDtmf?.Trim() ?? "";

    // ---------------------------------------------------------
    // Final lead status
    // ---------------------------------------------------------

    bool highIntentKey =
        temporaryDtmf is "1" or "2" or "3" or "4";

    string leadStatus;

    if (highIntentKey && duration >= 60)
    {
        leadStatus =
            $"Very Hot Lead - High Intent Key {temporaryDtmf} + Deep Engagement";
    }
    else if (highIntentKey && duration >= 30)
    {
        leadStatus =
            $"Hot Lead - High Intent Key {temporaryDtmf} + Good Engagement";
    }
    else if (highIntentKey)
    {
        leadStatus =
            $"Interested Lead - High Intent Key {temporaryDtmf}";
    }
    else if (!string.IsNullOrWhiteSpace(temporaryDtmf) &&
             duration >= 60)
    {
        leadStatus =
            $"Hot Lead - Deep IVR Engagement - Key {temporaryDtmf}";
    }
    else if (!string.IsNullOrWhiteSpace(temporaryDtmf) &&
             duration >= 30)
    {
        leadStatus =
            $"Warm Lead - IVR Exploration - Key {temporaryDtmf}";
    }
    else if (duration >= 60)
    {
        leadStatus =
            "Hot Lead - Long IVR Engagement";
    }
    else if (duration >= 30)
    {
        leadStatus =
            "Warm Lead - Good IVR Engagement";
    }
    else if (duration >= 15)
    {
        leadStatus =
            "Cold Lead - Short IVR Interaction";
    }
    else if (duration >= 10)
    {
        leadStatus =
            "Quick IVR Interaction";
    }
    else if (duration >= 5)
    {
        leadStatus =
            "Intro Drop";
    }
    else
    {
        leadStatus =
            "Immediate Drop";
    }

    // ---------------------------------------------------------
    // Find existing lead
    //
    // DB lookup happens only at HANGUP.
    // ---------------------------------------------------------

    var lead = await _context.LeadRecords
        .FirstOrDefaultAsync(
            x => x.Mobile == payload.Mobile,
            cancellationToken);

    // ---------------------------------------------------------
    // FINAL DB SAVE
    // ---------------------------------------------------------

    if (lead == null)
    {
        lead = new LeadRecord
        {
            Mobile = payload.Mobile!,
            LeadStatus = leadStatus,
            CallDuration = duration,

            PressedDtmf =
                string.IsNullOrWhiteSpace(temporaryDtmf)
                    ? null
                    : temporaryDtmf,

            LastEventType = "HANGUP",
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };

        _context.LeadRecords.Add(lead);
    }
    else
    {
        lead.LeadStatus = leadStatus;
        lead.CallDuration = duration;
        lead.LastEventType = "HANGUP";
        lead.UpdatedAt = DateTime.UtcNow;

        lead.PressedDtmf =
            string.IsNullOrWhiteSpace(temporaryDtmf)
                ? null
                : temporaryDtmf;
    }

    // ---------------------------------------------------------
    // FINAL SAVE
    // ---------------------------------------------------------

    await _context.SaveChangesAsync(cancellationToken);

    // ---------------------------------------------------------
    // Remove temporary DTMF after successful final save
    // ---------------------------------------------------------

    _cache.Remove(cacheKey);

    // ---------------------------------------------------------
    // RESPONSE
    // ---------------------------------------------------------

    return Ok(new
    {
        status = "success",
        templateId = 4,
        eventType,
        pressedKey,
        effectivePressedKey = temporaryDtmf,
        duration,
        leadStatus,
        message = "HANGUP received. Final lead data saved."
    });
}
```
